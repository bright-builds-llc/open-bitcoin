pub const DEFAULT_KERNEL_CACHE_BYTES: u64 = 450 * 1024 * 1024;
pub const MIN_DBCACHE_BYTES: u64 = 4 * 1024 * 1024;
pub const COINS_DB_CACHE_CAP_BYTES: u64 = 8 * 1024 * 1024;
pub const fn default_coins_cache_byte_limit() -> u64 {
    let leftover = DEFAULT_KERNEL_CACHE_BYTES.saturating_sub(COINS_DB_CACHE_CAP_BYTES);
    if leftover > MIN_DBCACHE_BYTES { leftover } else { MIN_DBCACHE_BYTES }
} // 442 MiB

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ManagerReadiness {
    NotReady,
    ReadyToFlush,

    cache_byte_limit: u64,
    mempool_leftover_bytes: u64,
    next_write: FlushPolicyTime,
    memory_pressure: bool,
}

pub struct FlushExecution {
    pub decision: FlushDecision,
    pub wrote_coins: bool,
}

/// Injected persist sink so undo/index abort tests do not chmod a Fjall datadir.
pub trait FlushPersistSink {
    fn persist_block(&self, block: &Block) -> Result<(), StorageError>;
    fn persist_undo(&self, hash: BlockHash, undo: &BlockUndo) -> Result<(), StorageError>;
    fn persist_header_entries(&self, entries: &[HeaderEntry]) -> Result<(), StorageError>;
}

impl FlushPersistSink for FjallNodeStore {
    fn persist_block(&self, block: &Block) -> Result<(), StorageError> {
        self.save_block(block, PersistMode::Flush).map(|_| ())
    }
    fn persist_undo(&self, hash: BlockHash, undo: &BlockUndo) -> Result<(), StorageError> {
        self.save_undo(hash, undo, PersistMode::Flush)
    }
    fn persist_header_entries(&self, entries: &[HeaderEntry]) -> Result<(), StorageError> {
        if entries.is_empty() {
            return Ok(());
        }
        self.save_header_entries(entries, PersistMode::Flush)
    }
}

pub fn initialize(
    store: &FjallNodeStore,
    now: FlushPolicyTime,
    next_write: FlushPolicyTime,
    mempool_leftover_bytes: u64,
    memory_pressure: bool,
    disk_free_bytes: u64,
) -> Result<(FlushLifecycle, FjallCoinsView, CoinsCache<FjallCoinsView>), StorageError>;

impl FlushLifecycle {
    pub fn readiness(&self) -> ManagerReadiness;
    pub fn execute_flush<V: CoinsView, S: FlushPersistSink>(
        &mut self,
        sink: &S,
        cache: &mut CoinsCache<V>,
        mode: FlushMode,
        now: FlushPolicyTime,
        disk_free_bytes: u64,
        undo_window: &[(BlockHash, BlockUndo)],
        header_entries: &[HeaderEntry],
        block_payloads: &[Block],
    ) -> Result<FlushExecution, StorageError>;
    pub fn set_next_write(&mut self, next_write: FlushPolicyTime);
}

`initialize` order (D-11), store already schema-2 open:
1. `FjallCoinsView::from_store(store)`
2. `heads = view.head_blocks()`; `decide_recovery(heads.len())`
3. `InterruptedTwoHeads` → `replay_interrupted_flush(store, view)?` (Plan 03)
4. `InconsistentOtherCount` → Corruption fail-closed (D-09)
5. `CoinsCache::from_parent(view)` empty overlay
6. `ManagerReadiness::ReadyToFlush`
Do not report ReadyToFlush, do not call `decide_flush`, before step 6.

`execute_flush` (D-14, D-15):
1. If `readiness != ReadyToFlush` return error (`StorageError::Corruption` with detail `flush before ready` is fine; do not write).
2. Build `FlushPolicyInput { mode, cache_bytes: cache.estimated_cache_bytes(), cache_byte_limit: self.cache_byte_limit, mempool_leftover_bytes: self.mempool_leftover_bytes, now, next_write: self.next_write, memory_pressure: self.memory_pressure, disk_free_bytes, cache_entry_count: cache.cache_entry_count() }`.
3. `decision = decide_flush(input)`. Do not re-derive Flush vs Sync.
4. `None` → `{ decision, wrote_coins: false }`.
5. `RefuseDiskSpace` → return `Err` mapped from a coins `StorageError::BackendFailure` or a dedicated `StorageError` already used for disk refusal; **do not** write coins. Prefer `StorageError::Corruption { namespace: Coins, detail: "refuse disk space", action: Repair }` if no existing variant fits. Do not add a new production crate.
6. `Flush` / `Sync`:
   a. For each `block_payloads` entry: `sink.persist_block(block)?`
   b. For each `undo_window` entry: `sink.persist_undo(hash, undo)?`
   c. `sink.persist_header_entries(header_entries)?` (no-op when empty)
   d. Only then `cache.flush()` on `FlushDecision::Flush`, or `cache.sync()` on `FlushDecision::Sync`
   e. Coins persist mode: `PersistMode::Sync` when `mode == FlushMode::Always`, else `PersistMode::Flush`. If `V` is `FjallCoinsView`, use `batch_write_with_persist_mode` after collecting the dirty batch; if the cache `flush`/`sync` path cannot take a persist mode, call `flush`/`sync` then coins persist is already inside FjallCoinsView — in that case pass mode by using `batch_write_with_persist_mode` on a Fjall-only helper rather than re-choosing cache Flush vs Sync.
7. If a/b/c fails, return the error before step d. `wrote_coins` is false. B must not change. Do not chmod datadirs; abort tests inject `FlushPersistSink`.

`probe_disk_free_bytes(datadir: &Path) -> u64` lives in this module (unix `libc::statvfs` `f_bavail * f_frsize`; non-unix `u64::MAX`). Default `memory_pressure = false` when callers omit it (research A2).

Do not change `ManagedChainstate::persist` (still `save_snapshot`) — Plan 05 owns that cutover (D-18 timing).
