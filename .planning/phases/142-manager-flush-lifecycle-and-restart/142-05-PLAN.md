```rust
pub struct ManagedChainstate<S, V: CoinsView = MemoryCoinsView> {
    store: S,
    chainstate: Chainstate<V>,
    flush_lifecycle: FlushLifecycle,
}

impl FlushLifecycle {
    /// Memory `from_store` / tests only. Production Fjall uses `initialize` (Plan 04 / 06).
    pub fn ready(
        now: FlushPolicyTime,
        next_write: FlushPolicyTime,
        mempool_leftover_bytes: u64,
        memory_pressure: bool,
    ) -> Self;
}

impl<V: CoinsView> Chainstate<V> {
    pub fn coins_mut(&mut self) -> &mut CoinsCache<V>;
}

impl FlushPersistSink for MemoryChainstateStore { /* persist_block/headers no-op Ok; persist_undo → save_undo */ }

impl<S: ChainstateStore + FlushPersistSink, V: CoinsView> ManagedChainstate<S, V> {
    pub fn from_store(store: S) -> Self {
        let chainstate = store.load_snapshot().map(Chainstate::from_snapshot).unwrap_or_default();
        Self {
            store,
            chainstate,
            flush_lifecycle: FlushLifecycle::ready(
                FlushPolicyTime::from_unix_seconds(0),
                FlushPolicyTime::from_unix_seconds(0),
                0,
                false,
            ),
        }
    }

    fn flush_window(&self) -> (Vec<(BlockHash, BlockUndo)>, Vec<HeaderEntry>, Vec<Block>) {
        // undo: every (hash, undo) from chainstate.snapshot().undo_by_block
        // headers: HeaderEntry mapping already used by persist_progress / save_header_entries
        // blocks: empty — connect already save_block'd (research Pattern 3)
    }

    fn flush_with_mode(&mut self, mode: FlushMode, now: FlushPolicyTime, disk_free_bytes: u64) -> Result<FlushExecution, StorageError> {
        let (undo_window, header_entries, block_payloads) = self.flush_window();
        self.flush_lifecycle.execute_flush(
            &self.store,
            self.chainstate.coins_mut(),
            mode,
            now,
            disk_free_bytes,
            &undo_window,
            &header_entries,
            &block_payloads,
        )
    }

    fn persist(&mut self) {
        let _ = self.flush_with_mode(
            FlushMode::IfNeeded,
            FlushPolicyTime::from_unix_seconds(0),
            u64::MAX,
        );
        // RefuseDiskSpace / None: do not panic; wrote_coins stays false. Never save_snapshot.
    }
}
```
