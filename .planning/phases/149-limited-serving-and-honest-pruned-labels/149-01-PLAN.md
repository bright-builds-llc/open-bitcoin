```rust
// message.rs, beside NETWORK = Self(1 << 0) and WITNESS = Self(1 << 3)
pub const NETWORK_LIMITED: Self = Self(1 << 10);

// limited_serve.rs
pub const LIMITED_SERVE_RACE_BUFFER: u32 = 2;

pub fn advertised_service_flags(mode: PruneMode) -> ServiceFlags {
    if matches!(mode, PruneMode::Disabled) {
        return ServiceFlags::NETWORK | ServiceFlags::WITNESS;
    }
    ServiceFlags::NETWORK_LIMITED | ServiceFlags::WITNESS
}

pub fn block_request_exceeds_limited_serve_window(tip_height: u32, block_height: u32) -> bool {
    tip_height.saturating_sub(block_height) > MIN_BLOCKS_TO_KEEP + LIMITED_SERVE_RACE_BUFFER
}
```

`MIN_BLOCKS_TO_KEEP` stays `288` in `packages/open-bitcoin-chainstate/src/prune/range.rs`. Do not edit `height_inside_keep_window`. The comparison is greater-than against the sum `288 + 2`, so distance `290` is inside and distance `291` is outside. `saturating_sub` yields not-outside when `block_height > tip_height`.

`advertised_service_flags` reads only `PruneMode`. It must not take a `have_pruned` argument. `PruneMode::ManualOnly` and `PruneMode::Automatic { target_mib }` share the limited bits. `LocalPeerConfig::default` and `VersionMessage::default` stay `ServiceFlags::NETWORK | ServiceFlags::WITNESS`.
