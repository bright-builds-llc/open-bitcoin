impl<S: ChainstateStore, V: CoinsView> ManagedPeerNetwork<S, V> {
    pub fn set_prune_mode(&mut self, mode: PruneMode) {
        self.prune_mode = mode;
        let services = advertised_service_flags(mode);
        self.local_config.services = services;
        self.peer_manager.set_local_services(services);
    }

    pub fn set_serving_have_pruned(&mut self, have_pruned: bool) {
        self.serving_have_pruned = have_pruned;
    }

    pub fn local_services(&self) -> ServiceFlags {
        self.local_config.services
    }
}

// runtime_authority.rs — add inside the existing
// impl<S: ChainstateStore, V: CoinsView> ManagedNetworkHandle<S, V>,
// next to set_inbound_admission_policy. The handle wraps
// Arc<Mutex<ManagedPeerNetwork>> and locks with self.mutate.
// Plan 03 calls this method. The inner setter returns ().
pub fn set_serving_have_pruned(
    &mut self,
    have_pruned: bool,
) -> Result<(), ManagedNetworkAuthorityError> {
    self.mutate(|network| network.set_serving_have_pruned(have_pruned))
}

// peer.rs
impl PeerManager {
    pub fn set_local_services(&mut self, services: ServiceFlags) {
        self.local_config.services = services;
    }
}
