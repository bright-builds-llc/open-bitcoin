```rust
// prune/mode.rs
pub const MIN_DISK_SPACE_FOR_BLOCK_FILES_MIB: u64 = 550;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PruneMode {
    Disabled,
    ManualOnly,
    Automatic { target_mib: u64 },
}

impl Default for PruneMode {
    fn default() -> Self { Self::Disabled }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PruneModeParseError {
    Negative,
    BelowMinimum { requested_mib: u64 },
}

pub fn parse_prune_arg(n_prune_arg: i64) -> Result<PruneMode, PruneModeParseError>;

// prune/range.rs
pub const MIN_BLOCKS_TO_KEEP: u32 = 288;

pub fn last_prunable_height(tip: u32) -> u32; // tip.saturating_sub(MIN_BLOCKS_TO_KEEP)

pub fn height_inside_keep_window(tip: u32, height: u32) -> bool;
// true iff height > tip.saturating_sub(MIN_BLOCKS_TO_KEEP)

pub fn automatic_prune_allowed(tip: u32, prune_after_height: u32) -> bool;
// tip > prune_after_height

// prune/locks.rs
pub const PRUNE_LOCK_BUFFER: u32 = 10;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PruneLockInfo {
    pub name: String,           // caller-supplied id; not persisted this phase
    pub height_first: u32,      // inclusive
    pub height_last: u32,       // inclusive
}

pub fn height_forbidden_by_lock(height: u32, lock: &PruneLockInfo) -> bool;
pub fn height_forbidden_by_any_lock(height: u32, locks: &[PruneLockInfo]) -> bool;
```

`height_forbidden_by_lock` MUST use the Knots formula (research §5), not naive ±10 both sides:

```text
lock_height = if height_first <= 11 { 1 } else { height_first - PRUNE_LOCK_BUFFER - 1 }
lock_height_last = height_last + PRUNE_LOCK_BUFFER
forbidden iff height > lock_height && height <= lock_height_last
```

For `height_first > 11`, that is inclusive `[height_first - 10, height_last + 10]`.
Fixture: lock `{first:100, last:100}` forbids heights `90..=110` inclusive.
